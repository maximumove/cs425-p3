## Design

IPv4 addresses are stored as 32-bit integers, and each CIDR block stores an
address and a prefix length. The subnet mask is formed from the leading
`prefix` one-bits. The `/0` case is handled separately and produces an all-zero
mask, avoiding a 32-bit shift; this makes `0.0.0.0/0` contain every address.
For `/32`, the mask is all ones, so the block contains exactly its one
specified address.

Host-address calculations follow the conventional network and broadcast
exclusions for prefixes `/0` through `/30`: the first and last host are the
network address plus one and broadcast address minus one, respectively, and
the host count is `2^(32 - prefix) - 2`. The `/0` count uses a 64-bit result
because `2^32` does not fit in 32 bits. `/31` is treated as a two-address
point-to-point block, so both addresses are usable rather than reserving
network and broadcast addresses. A `/32` is a single-address block, with that
address as its one usable host.

The forwarding table holds routes in a dynamically grown array and rejects
duplicate destination blocks. Lookup scans every route, uses the route's mask
to test whether it contains the destination, and keeps the match with the
greatest prefix length. Thus, more-specific routes win regardless of insertion
order; a `/0` default route is used only if no more-specific route matches.
Table-file entries must name a network address (host bits all zero), and
`default` is accepted as shorthand for `0.0.0.0/0`.

bool ip_parse(const char *text, uint32_t *addr)
{
  uint32_t value = 0;
  const char *cursor = text;

  if (text == NULL || addr == NULL) {
    return false;
  }

  for (unsigned int octet_index = 0; octet_index < 4; octet_index++) {
    unsigned int octet = 0;
    unsigned int digits = 0;

    if (!isdigit((unsigned char)*cursor)) {
      return false;
    }
    if (*cursor == '0' && isdigit((unsigned char)cursor[1])) {
      return false;
    }

    while (isdigit((unsigned char)*cursor)) {
      octet = octet * 10U + (unsigned int)(*cursor - '0');
      if (octet > 255U) {
        return false;
      }
      digits++;
      cursor++;
    }
    if (digits == 0U) {
      return false;
    }
    value = (value << 8) | (uint32_t)octet;

    if (octet_index < 3U) {
      if (*cursor != '.') {
        return false;
      }
      cursor++;
    } else if (*cursor != '\0') {
      return false;
    }
  }

  *addr = value;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  (void)snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
                 (unsigned int)((addr >> 24) & UINT32_C(0xff)),
                 (unsigned int)((addr >> 16) & UINT32_C(0xff)),
                 (unsigned int)((addr >> 8) & UINT32_C(0xff)),
                 (unsigned int)(addr & UINT32_C(0xff)));
}

bool cidr_parse(const char *text, cidr_t *block)
{
  char address_text[IP_STR_LEN];
  const char *slash;
  const char *prefix_text;
  size_t address_length;
  unsigned int prefix = 0;
  uint32_t address;
  cidr_t parsed;

  if (text == NULL || block == NULL) {
    return false;
  }
  slash = strchr(text, '/');
  if (slash == NULL || strchr(slash + 1, '/') != NULL) {
    return false;
  }
  address_length = (size_t)(slash - text);
  if (address_length == 0U || address_length >= sizeof(address_text)) {
    return false;
  }
  memcpy(address_text, text, address_length);
  address_text[address_length] = '\0';
  if (!ip_parse(address_text, &address)) {
    return false;
  }

  prefix_text = slash + 1;
  if (*prefix_text == '\0' || (*prefix_text == '0' && prefix_text[1] != '\0')) {
    return false;
  }
  for (const char *cursor = prefix_text; *cursor != '\0'; cursor++) {
    if (!isdigit((unsigned char)*cursor)) {
      return false;
    }
    prefix = prefix * 10U + (unsigned int)(*cursor - '0');
    if (prefix > 32U) {
      return false;
    }
  }

  parsed.addr = address;
  parsed.prefix = (uint8_t)prefix;
  *block = parsed;
  return true;
}

uint32_t cidr_mask(uint8_t prefix)
{
  if (prefix == 0U) {
    return 0U;
  }
  if (prefix >= 32U) {
    return UINT32_MAX;
  }
  return UINT32_MAX << (32U - prefix);
}

uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
  return block.addr | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block)
{
  uint8_t prefix = block.prefix > 32U ? 32U : block.prefix;
  uint32_t network = cidr_network(block);
  return prefix >= 31U ? network : network + 1U;
}

uint32_t cidr_last_host(cidr_t block)
{
  uint8_t prefix = block.prefix > 32U ? 32U : block.prefix;
  uint32_t broadcast = cidr_broadcast(block);
  return prefix >= 31U ? broadcast : broadcast - 1U;
}

uint64_t cidr_host_count(cidr_t block)
{
  uint8_t prefix = block.prefix > 32U ? 32U : block.prefix;
  if (prefix == 32U) {
    return UINT64_C(1);
  }
  if (prefix == 31U) {
    return UINT64_C(2);
  }
  return (UINT64_C(1) << (32U - prefix)) - UINT64_C(2);
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block)
{
  return cidr_network(block) == block.addr;

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  uint32_t remaining;
  unsigned int borrowed = 0;
  uint8_t prefix;
  uint64_t offset;
  cidr_t subnet;

  if (out == NULL || !cidr_is_network(block) || n == 0U || i >= n ||
      (n & (n - 1U)) != 0U) {
    return false;
  }
  remaining = n;
  while (remaining > 1U) {
    borrowed++;
    remaining >>= 1;
  }
  if ((unsigned int)block.prefix + borrowed > 32U) {
    return false;
  }

  prefix = (uint8_t)((unsigned int)block.prefix + borrowed);
  offset = (uint64_t)i << (32U - prefix);
  subnet.addr = block.addr + (uint32_t)offset;
  subnet.prefix = prefix;
  *out = subnet;
  return true;
}

static bool next_field(const char **cursor, const char **field, size_t *length)
{
  while (isspace((unsigned char)**cursor)) {
    (*cursor)++;
  }
  if (**cursor == '\0') {
    return false;
  }
  *field = *cursor;
  while (**cursor != '\0' && !isspace((unsigned char)**cursor)) {
    (*cursor)++;
  }
  *length = (size_t)(*cursor - *field);
  return true;
}

{
  return strlen(word) == length && memcmp(field, word, length) == 0;

static int read_line(FILE *in, char **line, size_t *capacity, size_t *length)
{
  int character;
  *length = 0U;

  while ((character = fgetc(in)) != EOF) {
    if (*length + 1U >= *capacity) {
      size_t new_capacity = *capacity == 0U ? 128U : *capacity * 2U;
      char *grown;
      if (new_capacity <= *capacity) {
        return -1;
      }
      grown = realloc(*line, new_capacity);
      if (grown == NULL) {
        return -1;
      }
      *line = grown;
      *capacity = new_capacity;
    }
    (*line)[(*length)++] = (char)character;
    if (character == '\n') {
      break;
    }
  }
  if (ferror(in)) {
    return -1;
  }
  if (*length == 0U) {
    return 0;
  }
  (*line)[*length] = '\0';
  return 1;
}

bool route_parse(const char *line, route_t *route)
{
  const char *cursor;
  const char *destination;
  const char *gateway;
  const char *device;
  size_t destination_length;
  size_t gateway_length;
  size_t device_length;
  size_t extra_length;
  const char *extra;
  char token[CIDR_STR_LEN];
  route_t parsed;

  if (line == NULL || route == NULL) {
    return false;
  }
  cursor = line;
  if (!next_field(&cursor, &destination, &destination_length) ||
      !next_field(&cursor, &gateway, &gateway_length) ||
      !next_field(&cursor, &device, &device_length) ||
      next_field(&cursor, &extra, &extra_length)) {
    return false;
  }
  if (destination_length >= sizeof(token) || device_length == 0U ||
      device_length >= DEV_NAME_LEN) {
    return false;
  }

  memcpy(token, destination, destination_length);
  token[destination_length] = '\0';
  if (field_equals(destination, destination_length, "default")) {
    parsed.dest.addr = 0U;
    parsed.dest.prefix = 0U;
  } else if (!cidr_parse(token, &parsed.dest) || !cidr_is_network(parsed.dest)) {
    return false;
  }

  if (field_equals(gateway, gateway_length, "direct")) {
    parsed.gateway = 0U;
  } else {
    if (gateway_length >= IP_STR_LEN) {
      return false;
    }
    memcpy(token, gateway, gateway_length);
    token[gateway_length] = '\0';
    if (!ip_parse(token, &parsed.gateway)) {
      return false;
    }
  }
  memcpy(parsed.dev, device, device_length);
  parsed.dev[device_length] = '\0';
  *route = parsed;
  return true;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
  route_t *grown;
  size_t new_capacity;

  if (table == NULL || route == NULL || !cidr_is_network(route->dest)) {
    return false;
  }
  for (size_t index = 0; index < table->count; index++) {
    if (table->routes[index].dest.prefix == route->dest.prefix &&
        table->routes[index].dest.addr == route->dest.addr) {
      return false;
    }
  }
  if (table->count == table->capacity) {
    if (table->capacity == 0U) {
      new_capacity = 4U;
    } else {
      if (table->capacity > SIZE_MAX / 2U) {
        return false;
      }
      new_capacity = table->capacity * 2U;
    }
    if (new_capacity > SIZE_MAX / sizeof(route_t)) {
      return false;
    }
    grown = realloc(table->routes, new_capacity * sizeof(route_t));
    if (grown == NULL) {
      return false;
    }
    table->routes = grown;
    table->capacity = new_capacity;
  }
  table->routes[table->count] = *route;
  table->count++;
  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
  char *line = NULL;
  size_t capacity = 0U;
  size_t line_number = 0U;
  size_t length;
  bool success = true;
  int line_status;

  if (bad_line != NULL) {
    *bad_line = 0U;
  }
  if (table == NULL || in == NULL) {
    return false;
  }

  while ((line_status = read_line(in, &line, &capacity, &length)) > 0) {
    char *comment;
    route_t route;
    bool blank = true;
    line_number++;
    comment = memchr(line, '#', length);
    if (comment != NULL) {
      *comment = '\0';
    }
    for (char *cursor = line; *cursor != '\0'; cursor++) {
      if (!isspace((unsigned char)*cursor)) {
        blank = false;
        break;
      }
    }
    if (blank) {
      continue;
    }
    if (!route_parse(line, &route) || !table_add(table, &route)) {
      if (bad_line != NULL) {
        *bad_line = line_number;
      }
      success = false;
      break;
    }
  }
  if (success && line_status < 0) {
    success = false;
    if (bad_line != NULL) {
      *bad_line = 0U;
    }
  }
  free(line);
  return success;
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
  const route_t *best = NULL;
  if (table == NULL) {
    return NULL;
  }
  for (size_t index = 0; index < table->count; index++) {
    const route_t *route = &table->routes[index];
    if (cidr_contains(route->dest, dest) &&
        (best == NULL || route->dest.prefix > best->dest.prefix)) {
      best = route;
    }
  }
  return best;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
  char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H

static void print_usage(FILE *stream)
{
  (void)fprintf(stream,
                "Usage: myapp info <address>/<prefix>\n"
                "       myapp split <network>/<prefix> <count>\n"
                "       myapp lookup <table-file> <address>...\n");

static bool parse_count(const char *text, uint32_t *count)
{
  uint32_t value = 0U;
  if (text == NULL || *text == '\0' || count == NULL) {
    return false;
  }
  for (const char *cursor = text; *cursor != '\0'; cursor++) {
    uint32_t digit;
    if (*cursor < '0' || *cursor > '9') {
      return false;
    }
    digit = (uint32_t)(*cursor - '0');
    if (value > (UINT32_MAX - digit) / 10U) {
      return false;
    }
    value = value * 10U + digit;
  }
  *count = value;
  return true;
}

static int run_info(const char *text)
{
  cidr_t block;
  char address[IP_STR_LEN];
  char mask[IP_STR_LEN];
  char network[IP_STR_LEN];
  char broadcast[IP_STR_LEN];
  char first[IP_STR_LEN];
  char last[IP_STR_LEN];

  if (!cidr_parse(text, &block)) {
    (void)fprintf(stderr, "myapp: not a valid CIDR address: %s\n", text);
    return 1;
  }
  ip_format(block.addr, address);
  ip_format(cidr_mask(block.prefix), mask);
  ip_format(cidr_network(block), network);
  ip_format(cidr_broadcast(block), broadcast);
  ip_format(cidr_first_host(block), first);
  ip_format(cidr_last_host(block), last);
  (void)printf("address    %s\n", address);
  (void)printf("prefix     /%u\n", (unsigned int)block.prefix);
  (void)printf("netmask    %s\n", mask);
  (void)printf("network    %s\n", network);
  (void)printf("broadcast  %s\n", broadcast);
  (void)printf("first host %s\n", first);
  (void)printf("last host  %s\n", last);
  (void)printf("hosts      %" PRIu64 "\n", cidr_host_count(block));
  return 0;
}

static int run_split(const char *text, const char *count_text)
{
  cidr_t block;
  uint32_t count;

  if (!cidr_parse(text, &block) || !cidr_is_network(block) ||
      !parse_count(count_text, &count) || count == 0U ||
      (count & (count - 1U)) != 0U) {
    (void)fprintf(stderr, "myapp: invalid network or subnet count\n");
    return 1;
  }
  for (uint64_t index = 0U; index < count; index++) {
    cidr_t subnet;
    char formatted[CIDR_STR_LEN];
    if (!cidr_subnet(block, count, (uint32_t)index, &subnet)) {
      (void)fprintf(stderr, "myapp: subnet count exceeds the block capacity\n");
      return 1;
    }
    cidr_format(subnet, formatted);
    (void)puts(formatted);
  }
  return 0;
}

static int run_lookup(const char *path, int address_count, char **addresses)
{
  FILE *input = fopen(path, "r");
  fwd_table_t *table;
  size_t bad_line;
  int exit_status = 0;

  if (input == NULL) {
    (void)fprintf(stderr, "myapp: %s: cannot open table file\n", path);
    return 2;
  }
  table = table_create();
  if (table == NULL) {
    (void)fclose(input);
    (void)fprintf(stderr, "myapp: could not allocate forwarding table\n");
    return 2;
  }
  if (!table_load(table, input, &bad_line)) {
    if (bad_line == 0U) {
      (void)fprintf(stderr, "myapp: %s: could not read table file\n", path);
    } else {
      (void)fprintf(stderr, "myapp: %s:%zu: not a valid route\n", path, bad_line);
    }
    (void)fclose(input);
    table_destroy(table);
    return 2;
  }
  (void)fclose(input);

  for (int index = 0; index < address_count; index++) {
    uint32_t address;
    const route_t *route;
    if (!ip_parse(addresses[index], &address)) {
      (void)fprintf(stderr, "myapp: not a valid address: %s\n", addresses[index]);
      exit_status = 1;
      continue;
    }
    route = table_lookup(table, address);
    if (route == NULL) {
      (void)printf("%s unreachable\n", addresses[index]);
      if (exit_status == 0) {
        exit_status = 3;
      }
    } else if (route->gateway != 0U) {
      char gateway[IP_STR_LEN];
      ip_format(route->gateway, gateway);
      (void)printf("%s via %s dev %s\n", addresses[index], gateway, route->dev);
    } else {
      (void)printf("%s dev %s\n", addresses[index], route->dev);
    }
  }
  table_destroy(table);
  return exit_status;
}

int main(int argc, char **argv)
{
  if (argc == 1) {
    print_usage(stdout);
    return 0;
  }
  if (strcmp(argv[1], "info") == 0 && argc == 3) {
    return run_info(argv[2]);
  }
  if (strcmp(argv[1], "split") == 0 && argc == 4) {
    return run_split(argv[2], argv[3]);
  }
  if (strcmp(argv[1], "lookup") == 0 && argc >= 4) {
    return run_lookup(argv[2], argc - 3, &argv[3]);
  }
  print_usage(stderr);
  return 1;
}
